let catalog=[],filter='All',cart=[],selectedVariant='',loading=true;let focusReturn;
const $=s=>document.querySelector(s),esc=s=>String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const money=n=>new Intl.NumberFormat('en-US',{style:'currency',currency:'USD'}).format(n);
const belongs=t=>String(t||'').toLowerCase().includes('charlotte storm aau');
const category=p=>/hood/i.test(p.title+' '+p.productType)?'Hoodies':/sweatshirt|crewneck/i.test(p.title+' '+p.productType)?'Sweatshirts':/tee|t-shirt|shirt|long sleeve/i.test(p.title+' '+p.productType)?'Tees':'Accessories';
const variant=id=>catalog.flatMap(p=>p.variants.map(v=>({...v,product:p}))).find(v=>v.id===id);
function openDialog(id){focusReturn=document.activeElement;$(id).showModal()}function closeDialog(id){$(id).close();focusReturn?.focus()}
async function loadCatalog(){
 $('#products').innerHTML='<p class="empty" role="status">Loading Charlotte Storm AAU products…</p>';
 try{const r=await fetch('/api/products',{cache:'no-store'}),d=await r.json();if(!r.ok)throw Error(d.error||'Unable to load products');
 catalog=(d.products||[]).filter(p=>belongs(p.title)).map(p=>({...p,name:p.title,category:category(p),variants:p.variants.nodes,price:Math.min(...p.variants.nodes.map(v=>Number(v.price.amount))),image:p.featuredImage?.url||''}));
 try{cart=JSON.parse(localStorage.getItem('storm-shopify-cart-v1')||'[]').filter(x=>variant(x.id)?.availableForSale&&Number.isInteger(x.qty)&&x.qty>0&&x.qty<=99)}catch{cart=[]}
 loading=false;render();saveCart();
 }catch(e){loading=false;$('#results').textContent='Products unavailable';$('#products').innerHTML='<p class="empty">'+esc(e.message)+' <button id="retryCatalog">Try again</button></p>';$('#retryCatalog').onclick=loadCatalog;}
}
function render(){
 if(loading)return;
 let list=catalog.filter(p=>(filter==='All'||p.category===filter)&&p.name.toLowerCase().includes($('#search').value.toLowerCase()));if($('#sort').value==='low')list.sort((a,b)=>a.price-b.price);if($('#sort').value==='high')list.sort((a,b)=>b.price-a.price);
 $('#categoryTitle').textContent=filter==='All'?'STORM ESSENTIALS':'STORM '+filter.toUpperCase();$('#results').textContent=`${list.length} product${list.length===1?'':'s'}`;document.querySelectorAll('.filters button').forEach(b=>b.classList.toggle('active',b.dataset.filter===filter));
 $('#products').innerHTML=list.length?list.map(p=>`<button class="product" data-product="${esc(p.id)}" aria-label="View ${esc(p.name)}"><div class="productPhoto">${p.image?`<img src="${esc(p.image)}" alt="${esc(p.name)}" loading="lazy">`:''}<span class="badge">${p.availableForSale?'CHARLOTTE STORM AAU':'SOLD OUT'}</span><span class="quick">VIEW PRODUCT</span></div><div class="productInfo"><span>${esc(p.name)}</span><span>${money(p.price)}</span></div><div class="productCategory">${p.category}</div></button>`).join(''):'<p class="empty">No Charlotte Storm AAU products match this selection.</p>';document.querySelectorAll('[data-product]').forEach(b=>b.onclick=()=>product(b.dataset.product));
}
function product(id){
 const p=catalog.find(x=>x.id===id);if(!p||!belongs(p.title))return;selectedVariant=(p.variants.find(v=>v.availableForSale)||p.variants[0])?.id;
 $('#productDetails').innerHTML=`<div class="detail">${p.image?`<img src="${esc(p.image)}" alt="${esc(p.name)}">`:''}<div><p class="eyebrow">CHARLOTTE STORM AAU / ${p.category.toUpperCase()}</p><h2>${esc(p.name)}</h2><p class="price" id="variantPrice"></p><p>${esc(p.description)}</p><label class="optionLabel" for="variantSelect">OPTIONS</label><select id="variantSelect" style="max-width:100%;padding:12px;margin:12px 0 24px">${p.variants.map(v=>`<option value="${esc(v.id)}" ${v.id===selectedVariant?'selected':''} ${!v.availableForSale?'disabled':''}>${esc(v.title)}${v.availableForSale?'':' — Sold out'}</option>`).join('')}</select><button class="primary" id="add">ADD TO BAG</button><p class="sampleNote">Secure checkout with GOAT Sports.</p></div></div>`;
 const update=()=>{selectedVariant=$('#variantSelect').value;const v=variant(selectedVariant);$('#variantPrice').textContent=money(Number(v.price.amount));$('#add').disabled=!v.availableForSale;$('#add').textContent=v.availableForSale?'ADD TO BAG':'SOLD OUT'};$('#variantSelect').onchange=update;update();
 $('#add').onclick=()=>{const v=variant(selectedVariant);if(!v?.availableForSale)return;const item=cart.find(x=>x.id===v.id);if(item){if(item.qty>=99)return;item.qty++}else cart.push({id:v.id,qty:1});saveCart();closeDialog('#productDialog');toast('Added to your bag')};openDialog('#productDialog');
}
function saveCart(){try{localStorage.setItem('storm-shopify-cart-v1',JSON.stringify(cart))}catch{}$('#count').textContent=cart.reduce((a,x)=>a+x.qty,0);renderCart()}
function renderCart(){
 $('#bagCount').textContent=`(${cart.reduce((a,x)=>a+x.qty,0)})`;
 $('#cartItems').innerHTML=cart.length?cart.map((x,i)=>{const v=variant(x.id),p=v.product;return `<div class="cartItem">${p.image?`<img src="${esc(p.image)}" alt="${esc(p.name)}">`:''}<div><strong>${esc(p.name)}</strong><p>${esc(v.title)} · ${money(Number(v.price.amount))}</p><div class="qty"><button data-qty="${i}" data-step="-1" aria-label="Decrease quantity">−</button><span>${x.qty}</span><button data-qty="${i}" data-step="1" aria-label="Increase quantity">+</button><button class="remove" data-remove="${i}">Remove</button></div></div></div>`}).join(''):'<div class="empty">Your bag is waiting for some Storm.</div>';
 $('#cartSummary').innerHTML=cart.length?`<div class="total"><span>Subtotal</span><span>${money(cart.reduce((a,x)=>a+Number(variant(x.id).price.amount)*x.qty,0))}</span></div><p>Shipping and taxes calculated at checkout.</p><button class="primary" id="checkout">SECURE CHECKOUT</button><p id="checkoutError" role="alert"></p>`:'';
 document.querySelectorAll('[data-qty]').forEach(b=>b.onclick=()=>{cart[+b.dataset.qty].qty=Math.min(99,cart[+b.dataset.qty].qty+ +b.dataset.step);cart=cart.filter(x=>x.qty>0);saveCart()});document.querySelectorAll('[data-remove]').forEach(b=>b.onclick=()=>{cart.splice(+b.dataset.remove,1);saveCart()});if($('#checkout'))$('#checkout').onclick=checkout;
}
async function checkout(){const button=$('#checkout');button.disabled=true;button.textContent='OPENING CHECKOUT…';try{const r=await fetch('/api/checkout',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({lines:cart.map(x=>({merchandiseId:x.id,quantity:x.qty}))})}),d=await r.json();if(!r.ok)throw Error(d.error||'Checkout unavailable');location.assign(d.checkoutUrl)}catch(e){$('#checkoutError').textContent=e.message;button.disabled=false;button.textContent='SECURE CHECKOUT'}}
let timer;function toast(msg){$('#toast').textContent=msg;$('#toast').classList.add('show');clearTimeout(timer);timer=setTimeout(()=>$('#toast').classList.remove('show'),2400)}
document.querySelectorAll('[data-filter]').forEach(b=>b.onclick=()=>{filter=b.dataset.filter;$('#search').value='';render();$('#shop').scrollIntoView()});document.querySelectorAll('[data-shop]').forEach(b=>b.onclick=()=>{filter='All';$('#search').value='';render();$('#shop').scrollIntoView()});document.querySelectorAll('[data-close]').forEach(b=>b.onclick=()=>closeDialog('#'+b.dataset.close));document.querySelectorAll('dialog').forEach(d=>d.addEventListener('click',e=>{if(e.target===d){const r=d.getBoundingClientRect();if(e.clientX<r.left||e.clientX>r.right||e.clientY<r.top||e.clientY>r.bottom)closeDialog('#'+d.id)}}));$('#sort').onchange=render;$('#search').oninput=()=>{render();$('#shop').scrollIntoView()};$('#searchToggle').onclick=()=>{$('#searchBar').hidden=!$('#searchBar').hidden;if(!$('#searchBar').hidden)$('#search').focus()};$('#closeSearch').onclick=()=>{$('#searchBar').hidden=true;$('#search').value='';render();$('#searchToggle').focus()};$('#cartToggle').onclick=()=>{renderCart();openDialog('#cartDialog')};loadCatalog();
